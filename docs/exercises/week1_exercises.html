Exercise 1 — apply() & which()

The matrix M below has 5 rows and 4 columns. Work through the following tasks:

  1. Calculate the mean of each row using apply().
  2. Calculate the mean of each column using apply().
  3. Find the positions (indices) of all positive elements (> 0) using which().
  4. Get the actual values of all positive elements.
#| '!! shinylive warning !!': |
#|   shinylive does not work in self-contained HTML documents.
#|   Please set `embed-resources: false` in your metadata.
#| standalone: true
#| viewerHeight: 260
library(shiny)
ui <- fluidPage(
  textAreaInput("code", NULL, value = "M <- matrix(c(-2,3,1,-5,4,2,0,-1,3,6,-3,2,1,4,-2,5,2,-4,3,1), nrow = 5)\n", width = '100%', height = '160px'),
  actionButton("run", "Run Code", class = "btn-success"),
  verbatimTextOutput("out")
)
server <- function(input, output) {
  observeEvent(input$run, {
    output$out <- renderPrint({
      exprs <- parse(text = input$code)
      for (e in exprs) { res <- withVisible(eval(e)); if (!is.null(res$value) && res$visible) print(res$value) }
    })
  })
}
shinyApp(ui, server)
# Task 1 – mean of each row:
apply(M, 1, mean)

# Task 2 – mean of each column:
apply(M, 2, mean)

# Task 3 – positions of positive elements:
which(M > 0)

# Task 4 – values of positive elements:
M[which(M > 0)]

Exercise 1 — numpy axis operations & np.where()

The matrix M below has 5 rows and 4 columns. Work through the following tasks:

  1. Calculate the mean of each row using np.mean() with axis=1.
  2. Calculate the mean of each column using np.mean() with axis=0.
  3. Find the positions of all positive elements (> 0) using np.where().
  4. Get the actual values of all positive elements.
#| '!! shinylive warning !!': |
#|   shinylive does not work in self-contained HTML documents.
#|   Please set `embed-resources: false` in your metadata.
#| standalone: true
#| viewerHeight: 270
from shiny import App, ui, render, reactive
import io, contextlib, ast
import numpy as np

app_ui = ui.page_fluid(
    ui.input_text_area("code", None, value="import numpy as np\nM = np.array([-2,3,1,-5,4,2,0,-1,3,6,-3,2,1,4,-2,5,2,-4,3,1], dtype=float).reshape(5,4)\n", width='100%', height='160px'),
    ui.input_action_button("run", "Run Code", class_="btn-success"),
    ui.output_text_verbatim("out")
)
def server(input, output, session):
    @render.text
    @reactive.event(input.run)
    async def out():
        f = io.StringIO(); ns = {"np": np}
        with contextlib.redirect_stdout(f):
            try:
                tree = ast.parse(input.code())
                for node in tree.body:
                    if isinstance(node, ast.Expr):
                        val = eval(compile(ast.Expression(node.value),'<s>','eval'), ns)
                        if val is not None: print(val)
                    else: exec(compile(ast.Module([node],[]),'<s>','exec'), ns)
            except Exception as e: print(f"Error: {e}")
        return f.getvalue()
app = App(app_ui, server)
# Task 1 – mean of each row:
print(np.mean(M, axis=1))

# Task 2 – mean of each column:
print(np.mean(M, axis=0))

# Task 3 – positions of positive elements:
print(np.where(M > 0))

# Task 4 – values of positive elements:
print(M[np.where(M > 0)])

Exercise 3 ⭐ Expert — map() + ternary expression

Given the list y = [1, 3, -5, 4, -10], use map() with a lambda function to transform every element:

  • Return sqrt(x) if the element is positive (> 0).
  • Return 0 if the element is zero or negative.

Hint: list(map(lambda x: ... , y))

#| '!! shinylive warning !!': |
#|   shinylive does not work in self-contained HTML documents.

from shiny import App, ui, render, reactive
import io, contextlib, ast
import math

app_ui = ui.page_fluid(
    ui.input_text_area("code", None, value="import math\ny = [1, 3, -5, 4, -10]\n", width='100%', height='130px'),
    ui.input_action_button("run", "Run Code", class_="btn-success"),
    ui.output_text_verbatim("out")
)
def server(input, output, session):
    @render.text
    @reactive.event(input.run)
    async def out():
        f = io.StringIO(); ns = {"math": math}
        with contextlib.redirect_stdout(f):
            try:
                tree = ast.parse(input.code())
                for node in tree.body:
                    if isinstance(node, ast.Expr):
                        val = eval(compile(ast.Expression(node.value),'<s>','eval'), ns)
                        if val is not None: print(val)
                    else: exec(compile(ast.Module([node],[]),'<s>','exec'), ns)
            except Exception as e: print(f"Error: {e}")
