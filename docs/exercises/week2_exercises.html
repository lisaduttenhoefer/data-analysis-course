# Simulate expression matrix (negative binomial)
expr_mat <- matrix(0, nrow = n_genes, ncol = n_samples)

  expr_mat[i, ] <- rnbinom(n_samples, mu = gene_means[i], size = 5)

mean_expr <- rowMeans(expr_mat)
var_expr  <- apply(expr_mat, 1, var)

plot(mean_expr, var_expr,
     xlab = "Mean Expression", ylab = "Variance",
     main = "Mean-Variance Relationship in Gene Expression",
     pch = 16, col = "blue")

abline(lm(var_expr ~ mean_expr), col = "green", lwd = 2)   # actual trend

# Simulate expression matrix (negative binomial)
size = 5
expr_mat = np.zeros((n_genes, n_samples))
for i in range(n_genes):
    mu = gene_means[i]
    p  = size / (size + mu)
    expr_mat[i, :] = np.random.negative_binomial(size, p, n_samples)

plt.scatter(mean_expr, var_expr, color='blue', alpha=0.6, s=20)
plt.plot([0, mean_expr.max()], [0, mean_expr.max()],
         'r--', lw=2, label='Poisson (Var = Mean)')
m, b = np.polyfit(mean_expr, var_expr, 1)
x_fit = np.linspace(0, mean_expr.max(), 200)
plt.plot(x_fit, m*x_fit + b, 'g-', lw=2, label='Actual trend')

plt.ylabel('Variance')
plt.title('Mean–Variance Relationship in Gene Expression')
